cmake_minimum_required(VERSION 3.22)
project(NMG VERSION 1.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(MSVC)
 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(NMG_PLATFORM_OPTIONS)
if(WIN32)
 list(APPEND NMG_PLATFORM_OPTIONS NEEDS_WEBVIEW2 TRUE)
endif()
if(APPLE)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "12.0" CACHE STRING "Minimum macOS")
endif()
add_subdirectory(vendor/JUCE)
set(NMG_FORMATS VST3 CACHE STRING "Plugin formats; no standalone target")
juce_add_binary_data(NMGWeb SOURCES Web/index.html)
juce_add_plugin(NMG ${NMG_PLATFORM_OPTIONS} COMPANY_NAME "NMG" BUNDLE_ID "com.nmg.numerical" PLUGIN_MANUFACTURER_CODE NmGc PLUGIN_CODE NmG1 PRODUCT_NAME "NMG" FORMATS ${NMG_FORMATS} IS_SYNTH TRUE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT TRUE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Instrument Generator)
juce_generate_juce_header(NMG)
target_sources(NMG PRIVATE Source/Plugin.cpp Source/Plugin.h Source/Scheduler.h)
if(APPLE)
 target_sources(NMG PRIVATE Source/MidiDrag.mm)
endif()
target_compile_definitions(NMG PUBLIC JUCE_WEB_BROWSER=1 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_REPORT_APP_USAGE=0)
if(WIN32)
 target_compile_definitions(NMG PUBLIC JUCE_USE_WIN_WEBVIEW2_WITH_STATIC_LINKING=1)
endif()
target_link_libraries(NMG PRIVATE NMGWeb juce::juce_audio_utils juce::juce_gui_extra PUBLIC juce::juce_recommended_config_flags)
add_executable(nmg_scheduler_tests Tests/scheduler.cpp)
target_include_directories(nmg_scheduler_tests PRIVATE Source)
enable_testing()
add_test(NAME scheduler COMMAND nmg_scheduler_tests)
# This is a headless plugin test executable, not a standalone user application.
add_executable(NMGNativeTests)
target_sources(NMGNativeTests PRIVATE Tests/native.cpp)
target_include_directories(NMGNativeTests PRIVATE Source $<TARGET_PROPERTY:NMG,INCLUDE_DIRECTORIES>)
target_link_libraries(NMGNativeTests PRIVATE NMG)
add_test(NAME native_state COMMAND NMGNativeTests)
add_executable(NMGEditorTests Tests/editor.cpp)
target_include_directories(NMGEditorTests PRIVATE Source $<TARGET_PROPERTY:NMG,INCLUDE_DIRECTORIES>)
target_link_libraries(NMGEditorTests PRIVATE NMG)
if(APPLE)
 target_sources(NMGEditorTests PRIVATE Tests/snapshot.mm)
endif()
# Editor test opens its own temporary WKWebView; no FL Studio automation.
# Loads the compiled VST3 dynamically, independently of the plugin's shared code.
if(APPLE)
 add_executable(NMGVST3Tests Tests/vst3.cpp Tests/vst3_web.mm Tests/snapshot.mm)
 target_compile_definitions(NMGVST3Tests PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=1 JUCE_USE_CURL=0)
 target_link_libraries(NMGVST3Tests PRIVATE juce::juce_audio_utils juce::juce_gui_extra)
endif()
# Sign after JUCE has generated VST3 moduleinfo.json (and all bundle resources).
if(APPLE)
 foreach(format VST3 AU)
  if(TARGET NMG_${format})
   add_custom_command(TARGET NMG_${format} POST_BUILD
    COMMAND /usr/bin/codesign --force --deep --sign - "$<TARGET_BUNDLE_DIR:NMG_${format}>")
  endif()
 endforeach()
endif()
